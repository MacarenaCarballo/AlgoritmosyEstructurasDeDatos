int procesarArchivos(char *nombreArch, tArbol *pa, tCmp comparar)
{
    char palabraBuffer[MAXLINEA+1];
    char palabra[MAXLINEA+1];
    FILE *pf = fopen(nombreArch, "rt");
    char *palabraGuardar;
    char *letra;
    int palabraEnProceso = 0;

    if(!pf)
        return ERRORCOD;

    palabraGuardar = palabra;

    while(fgets(palabraBuffer, MAXLINEA+1, pf))
    {
       // printf("%s\n", palabraBuffer);
        letra = palabraBuffer;

        while(*letra != '\0')
        {
            if(ES_CARACTER(*letra))
            {
                *palabraGuardar = *letra;
                palabraGuardar++;
                palabraEnProceso = 1;
            }
            else
            {
                if(palabraEnProceso && *letra!='\n')
                {
                    *palabraGuardar = '\0';
                    printf("%s\n", palabra);
                    insertarEnArbol(pa,&palabra,sizeof(char)*strlen(palabra),comparar);
                    palabraGuardar = palabra;
                    palabraEnProceso = 0;
                }
            }
            letra++;
        }
    }

    if(palabraEnProceso)
    {
        *palabraGuardar = '\0';
        printf("%s\n", palabra);
        insertarEnArbol(pa,&palabra,sizeof(char)*strlen(palabra),comparar);
    }

    fclose(pf);
    return REALIZADO;
}
